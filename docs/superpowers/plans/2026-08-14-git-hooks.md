# git hooks (pre-push 커밋 자동 등록) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `ontime hooks install` 한 번으로 어떤 git repo에서든 push 시 새 커밋들이 자동으로 OnTime API에 등록되게 한다.

**Architecture:** 새 모듈 `src/ontime_cli/hooks.py`에 typer 서브앱 `hooks_app`을 만든다. 설치되는 `.git/hooks/pre-push` 스크립트는 `ontime hooks pre-push "$@"`를 호출하는 껍데기(항상 exit 0)이고, stdin 파싱·rev-list·API 호출 로직은 전부 Python에 둔다. 기존 `ApiClient` 재사용.

**Tech Stack:** Python ≥ 3.9, typer, httpx (기존 의존성 그대로 — 신규 의존성 없음), 테스트는 pytest + typer의 `CliRunner`.

## Global Constraints

- Python ≥ 3.9 문법만 사용 (스펙: 런타임 Python ≥ 3.9)
- 신규 의존성 추가 금지 (스펙: 의존성은 typer/httpx 2개)
- 훅은 어떤 경우에도 push를 막지 않는다 (등록 실패 = 경고 후 정상 진행)
- 사용자 대상 메시지는 기존 코드처럼 한국어, 훅 출력은 `[ontime] ` 접두어

from __future__ import annotations

import typer

hooks_app = typer.Typer(help="git hooks 관리", no_args_is_help=True)


def repo_from_url(url: str) -> str:
    name = url.rstrip("/").split("/")[-1]
    if ":" in name:  # scp 형식 git@host:repo.git
        name = name.split(":")[-1]
    if name.endswith(".git"):
        name = name[: -len(".git")]
    return name


def parse_push_lines(lines: list) -> list:
    result = []
    for line in lines:
        parts = line.split()
        if len(parts) != 4:
            continue
        local_ref, local_sha, _remote_ref, remote_sha = parts
        if set(local_sha) == {"0"}:  # 브랜치 삭제 push
            continue
        branch = local_ref[len("refs/heads/"):] if local_ref.startswith("refs/heads/") else local_ref
        result.append((branch, local_sha, remote_sha))
    return result
```

import subprocess


def _git(*args: str) -> str:
    out = subprocess.run(["git", *args], capture_output=True, text=True, check=True)
    return out.stdout


def commits_to_push(local_sha: str, remote_sha: str, remote: str) -> list:
    if set(remote_sha) == {"0"}:  # 신규 브랜치: 원격에 없는 커밋 전부
        args = ["rev-list", local_sha, "--not", f"--remotes={remote}"]
    else:
        args = ["rev-list", f"{remote_sha}..{local_sha}"]
    return _git(*args).split()


def commit_info(sha: str) -> tuple:
    author, subject = _git("log", "-1", "--format=%an%x00%s", sha).rstrip("\n").split("\x00", 1)
    return author, subject
```

```python
import sys

from .api import ApiClient, ApiError


@hooks_app.command("pre-push", hidden=True)
def pre_push(remote: str = typer.Argument(...), url: str = typer.Argument(...)) -> None:
    """(내부용) git pre-push 훅 진입점. stdin 으로 push ref 줄들을 받는다."""
    repo = repo_from_url(url)
    try:
        client = ApiClient()
    except Exception as e:
        typer.echo(f"[ontime] 커밋 등록 건너뜀: {e}", err=True)
        return
    ok = failed = 0
    for branch, local_sha, remote_sha in parse_push_lines(sys.stdin.read().splitlines()):
        for sha in commits_to_push(local_sha, remote_sha, remote):
            author, subject = commit_info(sha)
            try:
                client.request(
                    "POST",
                    "/v1/developers/commits",
                    json={
                        "commitId": sha,
                        "repo": repo,
                        "branch": branch,
                        "author": author,
                        "commitMessage": subject,
                    },
                )
                ok += 1
            except Exception as e:
                typer.echo(f"[ontime] {sha[:8]} 등록 실패: {e}", err=True)
                failed += 1
    message = f"[ontime] {ok}개 커밋 등록 완료"
    if failed:
        message += f" ({failed}개 실패)"
    typer.echo(message)
```

```python
from pathlib import Path

MARKER = "# ontime-developer-cli pre-push hook"
HOOK_SCRIPT = f"""#!/bin/sh
{MARKER}
ontime hooks pre-push "$@" || echo "[ontime] 커밋 등록 실패 (push는 계속 진행)" >&2
exit 0
"""


def _hook_path() -> Path:
    out = subprocess.run(
        ["git", "rev-parse", "--git-path", "hooks/pre-push"],
        capture_output=True, text=True,
    )
    if out.returncode != 0:
        typer.echo("git 저장소가 아닙니다.", err=True)
        raise typer.Exit(1)
    return Path(out.stdout.strip())


@hooks_app.command()
def install(
    force: bool = typer.Option(False, "--force", help="기존 pre-push 훅 덮어쓰기"),
) -> None:
    """현재 repo 에 pre-push 커밋 자동 등록 훅을 설치한다."""
    hook = _hook_path()
    if hook.exists() and MARKER not in hook.read_text() and not force:
        typer.echo(
            "다른 pre-push 훅이 이미 있습니다. --force 로 덮어쓸 수 있습니다.", err=True
        )
        raise typer.Exit(1)
    hook.parent.mkdir(parents=True, exist_ok=True)
    hook.write_text(HOOK_SCRIPT)
    hook.chmod(0o755)
    typer.echo(f"pre-push 훅 설치 완료: {hook}")


@hooks_app.command()
def uninstall() -> None:
    """설치된 ontime pre-push 훅을 삭제한다."""
    hook = _hook_path()
    if not hook.exists() or MARKER not in hook.read_text():
        typer.echo("설치된 ontime pre-push 훅이 없습니다.")
        return
    hook.unlink()
    typer.echo("pre-push 훅 삭제 완료.")
```
